import os
import re
import io
import openpyxl
import copy

# PDF 처리 라이브러리 (uv add pypdf)
from pypdf import PdfReader, PdfWriter

# 텍스트(라벨) 생성을 위한 라이브러리 (uv add reportlab)
from reportlab.pdfgen import canvas
from reportlab.pdfbase import pdfmetrics
from reportlab.pdfbase.ttfonts import TTFont

# ============================================================
# 1) 경로 및 파일 설정
# ============================================================
BASE_DIR = r"D:\mvp_2"
SOURCE_DIR = os.path.join(BASE_DIR, "source")
INDEX_FILE = os.path.join(BASE_DIR, "pages.xlsx")

SOURCE_FILES = {
    "C": "1_사업계획서.pdf",
    "D": "2_보고서.pdf",
    "E": "3_지침서.pdf",
}

TOPIC_FOLDERS = {
    2: "a_학급수 및 학생수",
    3: "b_각종 면적",
    4: "c_사업 유형",
    5: "d_사업 기간",
    6: "e_설계발주방식",
    7: "f_사업비 내역",
    8: "g_스페이스 프로그램",
    9: "h_설문조사 결과",
}


# ============================================================
# 2) 한글 폰트 등록 (ReportLab용)
# ============================================================
def register_korean_font():
    font_name = "MalgunGothic"
    font_paths = [
        r"C:\Windows\Fonts\malgun.ttf",
        r"C:\Windows\Fonts\gulim.ttc"
    ]
    for path in font_paths:
        if os.path.exists(path):
            try:
                pdfmetrics.registerFont(TTFont(font_name, path))
                return font_name
            except:
                continue
    return "Helvetica"


KOREAN_FONT = register_korean_font()


# ============================================================
# 3) 라벨(워터마크) 생성 함수
# ============================================================
def create_label_packet(text):
    packet = io.BytesIO()
    can = canvas.Canvas(packet)
    can.setFont(KOREAN_FONT, 10)
    can.setFillColorRGB(0, 0, 0)
    can.drawString(20, 20, text)  # 좌측 하단 출력
    can.save()
    packet.seek(0)
    return packet


# ============================================================
# 4) 유틸리티 함수 (엑셀 파싱 등)
# ============================================================
def parse_page_numbers(cell_value):
    if cell_value is None: return []
    s = str(cell_value).strip().replace("\n", ",").replace(" ", "").replace(";", ",")
    if s in ["", "-"]: return []

    pages = set()
    for part in [p for p in s.split(",") if p]:
        normalized = part.replace("–", "-").replace("~", "-")
        if "-" in normalized:
            try:
                a_str, b_str = normalized.split("-", 1)
                a, b = int(float(a_str)), int(float(b_str))
                if a > b: a, b = b, a
                pages.update(range(a, b + 1))
            except:
                continue
        else:
            try:
                pages.add(int(float(normalized)))
            except:
                continue
    return sorted(pages)


# ============================================================
# 5) 메인 실행 로직
# ============================================================
def process_extraction():
    # 폴더 생성
    for folder in TOPIC_FOLDERS.values():
        os.makedirs(os.path.join(BASE_DIR, folder), exist_ok=True)

    wb = openpyxl.load_workbook(INDEX_FILE, data_only=True, read_only=True)
    sheet = wb["Sheet1"]

    print("\n" + "=" * 60)
    print("PDF 페이지 추출 + 라벨링 + 테마별 병합 시작")
    print("=" * 60)

    total_extracted = 0
    readers = {}
    handles = {}

    try:
        for row_num, folder_name in TOPIC_FOLDERS.items():
            folder_path = os.path.join(BASE_DIR, folder_name)
            print(f"\n📂 [{folder_name}] 처리 중...")

            label_packet = create_label_packet(folder_name)
            label_page = PdfReader(label_packet).pages[0]
            merged_writer = PdfWriter()
            topic_has_pages = False

            for col_letter, source_filename in SOURCE_FILES.items():
                target_pages = parse_page_numbers(sheet[f"{col_letter}{row_num}"].value)
                if not target_pages: continue

                # Reader 캐싱
                if source_filename not in readers:
                    f = open(os.path.join(SOURCE_DIR, source_filename), "rb")
                    handles[source_filename] = f
                    readers[source_filename] = PdfReader(f)

                reader = readers[source_filename]
                source_basename = os.path.splitext(source_filename)[0]

                for p_num in target_pages:
                    if p_num < 1 or p_num > len(reader.pages): continue

                    # 페이지 복제 및 라벨링 (원본 오염 방지)
                    working_page = copy.copy(reader.pages[p_num - 1])
                    working_page.merge_page(label_page)

                    # 1. 개별 파일 저장
                    single_writer = PdfWriter()
                    single_writer.add_page(working_page)
                    out_path = os.path.join(folder_path, f"{source_basename}__p.{p_num}__{folder_name}.pdf")
                    with open(out_path, "wb") as f_out:
                        single_writer.write(f_out)

                    # 2. 통합 파일용 추가
                    merged_writer.add_page(working_page)
                    topic_has_pages = True
                    total_extracted += 1



            if topic_has_pages:
                with open(os.path.join(BASE_DIR, f"{folder_name}.pdf"), "wb") as f_merged:
                    merged_writer.write(f_merged)
                print(f"  ⭐ 통합 저장 완료: {folder_name}.pdf")

        # [최종 마스터 통합본 생성 추가] --------------------------------------------
        print("\n" + "★" * 60)
        print("모든 테마 PDF를 순서대로 합쳐 '통합본.pdf'를 생성합니다.")

        master_merger = PdfWriter()
        # TOPIC_FOLDERS의 폴더명들을 이름순(a, b, c...)으로 정렬
        sorted_topic_files = sorted([f"{name}.pdf" for name in TOPIC_FOLDERS.values()])

        topic_count = 0
        for pdf_name in sorted_topic_files:
            pdf_path = os.path.join(BASE_DIR, pdf_name)

            if os.path.exists(pdf_path):
                # 생성된 테마 PDF를 읽어서 마스터 병합기에 추가
                temp_reader = PdfReader(pdf_path)
                for page in temp_reader.pages:
                    master_merger.add_page(page)
                print(f"  + {pdf_name} 병합 추가 완료")
                topic_count += 1

        if topic_count > 0:
            master_final_path = os.path.join(BASE_DIR, "통합본.pdf")
            with open(master_final_path, "wb") as f_master:
                master_merger.write(f_master)
            print(f"📂 최종 통합본 저장 완료: {master_final_path}")
        # --------------------------------------------------------------------------

    finally:
        for f in handles.values(): f.close()
        print(f"\n✅ 작업 완료. 총 {total_extracted}장 추출됨.")


if __name__ == "__main__":
    process_extraction()
