# [스크립트 요약]
# 기능: 지정된 폴더 경로를 탐색하여 모든 한글 파일(.hwp, .hwpx)을 PDF로 자동 변환합니다.
# 특징: tqdm 라이브러리를 활용해 변환 진행률(Progress Bar)을 시각적으로 표시하며, 특정 파일 오류 시 중단 없이 다음 파일로 넘어갑니다.
# 필수사항: Windows OS 및 한컴오피스 설치 필수, 파이썬 라이브러리 설치 필요 (uv add pyhwpx tqdm)

import os
import glob
from tqdm import tqdm
from pyhwpx import Hwp


def convert_all_hwp_to_pdf(folder_path=None):
    """
    지정된 폴더 내의 모든 HWP, HWPX 파일을 PDF로 일괄 변환합니다.

    Args:
        folder_path (str): 한글 파일이 저장된 폴더 경로
    """

    # 1. 경로 유효성 검사
    if folder_path is None or not os.path.exists(folder_path):
        print(f"❌ 오류: 유효하지 않은 경로입니다. -> {folder_path}")
        return

    print(f"📂 '{folder_path}' 폴더 스캔 중...")

    # 2. 파일 리스트 수집 (대소문자 구분 없이 검색할 수 있도록 로직 보완 가능하나, 여기서는 기본 glob 사용)
    # 팁: glob은 윈도우에서 대소문자를 구분하지 않는 경우가 많지만, 명시적으로 둘 다 찾습니다.
    hwp_files = glob.glob(os.path.join(folder_path, "*.hwp"))
    hwpx_files = glob.glob(os.path.join(folder_path, "*.hwpx"))

    all_files = hwp_files + hwpx_files

    if not all_files:
        print("⚠️ 변환할 한글 파일(.hwp, .hwpx)이 없습니다.")
        return

    print(f"총 {len(all_files)}개의 파일을 발견했습니다. 변환을 시작합니다.")

    # 3. 한글 프로그램 백그라운드 실행
    # visible=True로 하면 진행 과정을 눈으로 볼 수 있어 디버깅에 좋습니다. (완성 후 False 권장)
    hwp = Hwp(visible=False)

    try:
        # 4. 변환 루프 (tqdm으로 진행률 표시)
        for file_path in tqdm(all_files, desc="Processing", unit="file"):
            try:
                # 파일 열기
                hwp.open(file_path)

                # PDF 파일명 생성 (확장자만 변경)
                base_name = os.path.splitext(file_path)[0]
                pdf_path = base_name + ".pdf"

                # PDF로 저장 (pyhwpx의 save_as 메서드 활용)
                hwp.save_as(pdf_path, format="PDF")

            except Exception as e:
                filename = os.path.basename(file_path)
                print(f"\n❌ [오류] {filename}: {e}")
                continue

    except Exception as main_e:
        print(f"🔥 치명적 오류 발생: {main_e}")

    finally:
        # 5. 한글 프로그램 종료 (반드시 실행되어야 함)
        hwp.quit()
        print("\n✅ 모든 작업이 완료되었습니다.")


# --- 실행부 ---
if __name__ == "__main__":

    target_folder = r"D:\mvp_1"

def encode_image(image_path):
    with open(image_path, "rb") as image_file:
        return base64.b64encode(image_file.read()).decode('utf-8')


def clean_text(text):
    """
    [진공청소기 기능 - 최종 완결판 V5]
    - Markdown 표(Table) 구조 완벽 보존
    - 짧은 구분선(---) 유지
    - 의미 없는 긴 구분선(----------------) 및 노이즈 제거
    """
    if not text: return ""

    # 1. HTML 태그 및 기본 공백 처리
    text = text.replace("<br>", "\n").replace("<br/>", "\n").replace("<br />", "\n")
    text = re.sub(r'[ \t]{20,}', ' ', text)  # 과도한 공백만 줄임

    lines = text.split('\n')
    cleaned_lines = []
    prev_line = None

    for line in lines:
        line = line.strip()
        if not line:
            continue

        # [판별 로직]
        # A. 텍스트 콘텐츠 존재 여부 (한글/영문/숫자)
        has_content = bool(re.search(r'[가-힣a-zA-Z0-9]', line))

        # B. 표(Table) 구조 여부 (파이프 '|'가 있으면 표로 간주)
        is_table_row = '|' in line

        # C. 무의미한 긴 구분선 여부
        # 파이프 없이 -, =, _, * 등의 특수문자만 10자 이상 반복되면 '쓰레기 줄'로 판단
        # 예: "-----------------------------------" -> True (삭제 대상)
        # 예: "| --- | --- |" -> False (표이므로 유지)
        # 예: "---" -> False (짧은 구분선이므로 유지)
        is_long_garbage = (not is_table_row) and bool(re.match(r'^[-=_*]{10,}$', line))

        # [삭제 조건]
        # 1. 무의미한 긴 구분선(is_long_garbage)인 경우 삭제
        if is_long_garbage:
            continue

        # 2. 내용도 없고(A X), 표도 아니고(B X), 짧은 구분선도 아닌 특수문자 찌꺼기인 경우 삭제
        # (단, 짧은 구분선 '---' 등은 has_content가 False라도 살려야 함)
        if not has_content and not is_table_row:
            # 순수 특수문자로만 구성된 짧은 줄(10자 미만)은 문단 구분용일 수 있으니 유지
            # 그 외 이상한 기호들만 있는 경우 삭제
            if not re.match(r'^[-=_*]{1,9}$', line):
                # 수학 기호 등이 단독으로 있을 수 있으므로,
                # 만약 수식이 중요하다면 이 부분 조건을 더 완화해야 합니다.
                # 현재는 일반적인 문서 기준으로 -, =, _, * 외의 특수문자만 있는 줄은 삭제합니다.
                pass

        # 3. 중복 라인 제거
        if line == prev_line:
            continue

        cleaned_lines.append(line)
        prev_line = line

    text = '\n\n'.join(cleaned_lines)
    return text.strip()

                    "3. **서식:** 제목 위계(#, ##)를 명확히 하고, HTML 태그(`<br>`)는 절대 사용하지 마.\n"
                    "4. **오탈자 교정:** 한글 문맥에 어색한 오탈자가 있다면 자연스럽게 교정해.\n"
                    "5. **순수 결과물:** '결과입니다' 같은 설명 없이 마크다운 텍스트만 출력해."
                )
            },
            {
                "type": "image_url",
                "image_url": f"data:image/png;base64,{base64_img}"
            }
        ]
    )

    max_retries = 5
    base_delay = 10

    for attempt in range(max_retries):
        try:
            response = model.invoke([message])

            # --- [추가된 방어 로직] 텍스트 안전 추출 ---
            raw_content = response.content
            if isinstance(raw_content, list):
                # 반환값이 리스트일 경우 내부 텍스트 요소만 뽑아서 하나로 결합
                raw_content = "".join([
                    item.get("text", "") if isinstance(item, dict) else str(item)
                    for item in raw_content
                ])
            elif not isinstance(raw_content, str):
                # 문자열도 리스트도 아닌 알 수 없는 객체일 경우 강제 문자열 변환
                raw_content = str(raw_content)
            # ----------------------------------------

            # ✅ 해결: 변환 완료된 안전한 문자열인 raw_content를 전달
            cleaned_content = clean_text(raw_content)

            logging.info(f"   📝 [Page {page_num}] 추출 성공 (글자 수: {len(cleaned_content)})")
            return cleaned_content

        except (ResourceExhausted, ServiceUnavailable, InternalServerError, DeadlineExceeded, Exception) as e:
            wait_time = base_delay * (2 ** attempt) + random.uniform(0, 5)
            logging.warning(
                f"   ⚠️ [Page {page_num}] 오류 발생({type(e).__name__}). {wait_time:.1f}초 후 재시도 ({attempt + 1}/{max_retries})")
            time.sleep(wait_time)

    return f"\n> **[Page {page_num} 분석 실패]**\n"


def process_single_page(page_data):
    """
    개별 페이지 처리 함수
    Args:
        page_data: (index, image_object, current_img_dir, current_txt_dir) 튜플
    """
    idx, page_image, current_img_dir, current_txt_dir = page_data
    page_num = idx + 1

    # 동적으로 전달받은 경로 사용
    img_path = current_img_dir / f"{page_num:03d}.png"
    txt_path = current_txt_dir / f"{page_num:03d}.txt"

    # [이어하기 기능]
    if txt_path.exists() and txt_path.stat().st_size > 50:
        with open(txt_path, "r", encoding="utf-8") as f:
            content = f.read()
            return (idx, content)

    try:
        page_image.save(img_path, "PNG")
        raw_text = analyze_image_with_retry(img_path, page_num)
        formatted_text = f"[[PAGE: {page_num}]]\n\n{raw_text}"

        with open(txt_path, "w", encoding="utf-8") as f:
            f.write(formatted_text)

        return (idx, formatted_text)

    except Exception as e:
        logging.error(f"   ❌ [Page {page_num}] 치명적 에러: {e}")
        error_text = f"<p.{page_num} start>\n\n(Error: {e})\n\n<p.{page_num} end>"
        return (idx, error_text)


def process_all_pdfs():
    # 1. PDF 파일 목록 스캔 및 정렬 (가나다순)
    pdf_files = sorted(list(BASE_DIR.glob("*.pdf")))

    if not pdf_files:
        logging.error("처리할 PDF 파일이 없습니다.")
        return

    logging.info(f"총 {len(pdf_files)}개의 PDF 파일을 발견했습니다. 순차 처리를 시작합니다.")

    # 2. 각 PDF 파일에 대해 반복 처리
    for file_idx, pdf_path in enumerate(pdf_files, 1):
        file_stem = pdf_path.stem  # 파일명 (확장자 제외)
        logging.info(f"\n{'=' * 50}")
        logging.info(f"📂 [{file_idx}/{len(pdf_files)}] 파일 처리 시작: {pdf_path.name}")
        logging.info(f"{'=' * 50}")

        # 3. 파일별 전용 폴더 경로 생성 (ex: file1_img, file1_txt)
        current_img_dir = BASE_DIR / f"{file_stem}_img"
        current_txt_dir = BASE_DIR / f"{file_stem}_txt"

        current_img_dir.mkdir(parents=True, exist_ok=True)
        current_txt_dir.mkdir(parents=True, exist_ok=True)

        logging.info(f">> [Step 1] PDF 이미지 변환 중... (DPI: 150)")

        try:
            pages = convert_from_path(pdf_path, 150, poppler_path=POPPLER_PATH)
        except Exception as e:
            logging.error(f"PDF 변환 실패 ({pdf_path.name}): {e}")
            continue  # 다음 파일로 넘어감

        total_pages = len(pages)
        logging.info(f">> [Step 2] 총 {total_pages}페이지 병렬 처리 시작")

        results_map = {}
        # 인자 패킹: (인덱스, 이미지, 이미지폴더경로, 텍스트폴더경로)
        page_items = [(i, page, current_img_dir, current_txt_dir) for i, page in enumerate(pages)]

        with ThreadPoolExecutor(max_workers=MAX_WORKERS) as executor:
            future_to_page = {
                executor.submit(process_single_page, item): item[0] + 1
                for item in page_items
            }

            while future_to_page:
                done, not_done = wait(future_to_page, timeout=10, return_when=FIRST_COMPLETED)

                for future in done:
                    try:
                        idx, text = future.result()
                        results_map[idx] = text
                    except Exception as e:
                        logging.error(f"Future 결과 처리 중 에러: {e}")
                    finally:
                        del future_to_page[future]

                if not_done:
                    running = sorted(future_to_page.values())
                    display = str(running[:8]) + ("..." if len(running) > 8 else "")
                    logging.info(f"   ⏳ [진행 중] 남은 페이지: {len(not_done)}개 / 현재 작업: {display}")

        # 4. 최종 병합 및 저장
        logging.info(">> [Step 3] 파일 병합 중...")
        sorted_results = [results_map[i] for i in range(total_pages) if i in results_map]

        final_content = "\n\n---\n\n".join(sorted_results)
        # 확장자를 .txt로 변경하여 저장
        final_save_path = BASE_DIR / f"{file_stem}.txt"

        with open(final_save_path, "w", encoding="utf-8") as f:
            f.write(final_content)

        logging.info(f"✅ [{file_idx}/{len(pdf_files)}] 완료! 저장위치: {final_save_path.name}")
        logging.info("-" * 30)

    logging.info("\n🎉 모든 PDF 파일 처리 완료!")


if __name__ == "__main__":
    start_time = time.time()
    process_all_pdfs()
    logging.info(f"전체 총 소요 시간: {time.time() - start_time:.2f}초")

[부록] poppler 문제 해결법



pdf2image 라이브러리는 단순히 Python 코드로 PDF를 읽는 게 아니라, 시스템에 설치된 'Poppler'라는 외부 엔진을 빌려와서 이미지를 그리는 방식입니다.
따라서 Python 패키지만 설치해서는 작동하지 않고, Poppler 라이브러리 바이너리(Binary) 파일이 윈도우 시스템에 실제로 존재해야 합니다. 

Poppler 설치 및 경로 확인

코드에 D:\Apps\poppler-25.12.0\Library\bin라고 경로를 지정하셨는데, 실제로 해당 폴더에 파일들이 있는지 확인이 필요합니다.
  • 설치 여부 확인: 위 경로로 이동했을 때 pdftocairo.exe 파일이 보이나요?
  • 파일이 없다면: Poppler for Windows (GitHub)에서 최신 버전(예: Release-25.12.0-0.zip)을 다운로드하세요.
  • 압축 해제: 다운로드한 파일을, 코드 내 폴더 구조에 맞게 압축을 풉니다.  (예: POPPLER_PATH = r"D:\Apps\poppler-25.12.0\Library\bin")
    • 또는 ...\Library\bin 이 들어간 폴더의 전체 경로를 복사하여, 코드 내의 POPPLER_PATH에 바꿔서 붙여넣으세요.

uv가 쓸데없이 파이썬 3.14 이상의 호환성까지 체크하지 않도록, 지원 버전을 명확히 제한해 주면 깔끔하게 해결됩니다.

    Ini, TOML
    requires-python = ">=3.10, <3.14"
  4. 다시 터미널에서 아래 명령어를 실행합니다.
