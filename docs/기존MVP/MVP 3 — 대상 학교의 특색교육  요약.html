3. 📝 코드 간략 소개

1) distill.py

  • 기능: 텍스트 파일들을 읽어 Gemini API를 통해 교육과정 핵심 내용을 추출하고, 최종적으로 학교 특색교육 리포트를 생성합니다.
  • 특징:
    • 2단계 프로세스: 개별 파일 요약(Step 1) 후 전체 데이터를 통합 분석(Step 2)하여 정확도를 높였습니다.
    • 페르소나 적용: 'Architect-Collaborator' 지침을 통해 건축적 서사와 교육적 비전이 융합된 고품질 답변을 유도합니다.
    • 환각 방지: 원문의 고유 명칭을 그대로 사용하도록 강제하는 프롬프트가 적용되었습니다.

2) env.py & .env

  • 기능: API 키와 같은 민감 정보를 코드와 분리하여 관리합니다.
  • 특징: os.getenv를 통해 안전하게 키를 불러오며, 키가 없을 경우 명확한 에러 메시지를 띄워줍니다.


4. 💡 실행 전 체크리스트

  1. 입력 데이터 확인: D:\mvp_3\input_data 폴더 안에 분석 대상이 되는 .txt 파일들이 들어있는지 확인하세요.
  2. API Key 확인: .env 파일에 유효한 Gemini API Key가 입력되어 있어야 합니다.
  3. 모델 명칭: 현재 gemini-2.0-flash-001로 설정되어 있습니다. 사용 가능한 모델 목록은 부록의 check_models.py로 확인 가능합니다.

import os
import logging
import time
from pathlib import Path

# [라이브러리] LangChain 및 Google GenAI 관련
from langchain_google_genai import ChatGoogleGenerativeAI
from langchain_core.messages import HumanMessage, SystemMessage
from google.api_core.exceptions import ResourceExhausted

# [설정] 로깅
logging.basicConfig(level=logging.INFO, format="%(asctime)s - %(levelname)s - %(message)s")

# env.py에서 API KEY 가져오기
from env import GEMINI_API_KEY

# ==========================================
# 1. 사용자 환경 및 인증 설정
# ==========================================
os.environ["GOOGLE_API_KEY"] = GEMINI_API_KEY
MODEL_NAME = "gemini-2.0-flash-001"

INPUT_DIR = Path(r"D:\mvp_3\input_data")
PROCESSED_DIR = Path(r"D:\mvp_3\input_processed")
OUTPUT_DIR = Path(r"D:\mvp_3\output")

TIMEOUT_SECONDS = 120
WAIT_TIME = 10  # API 호출 간격 (429 에러 방지용)

# ==========================================
# 2. 프롬프트 정의
# ==========================================
STEP1_SYSTEM_PROMPT = """
당신은 교육과정 문서 분석 전문가입니다. 
당신의 임무는 제공된 문서에서 '핵심 교육 활동, 특색 사업, 교육과정 운영 방식'을 추출하여 요약하는 것입니다.
[주의사항] 반드시 문서에 명시된 원래의 명칭을 그대로 사용하십시오.
"""

STEP2_SYSTEM_PROMPT = """
페르소나 호출: Architect-Collaborator
지금부터 당신은 'Architect-Collaborator'입니다. 교육 기획 및 공간 기획 전문가의 파트너로서 
데이터를 영감으로 전환하고, 프로젝트의 철학이 담긴 건축적 서사를 함께 구축합니다.
...(이하 페르소나 지침 상세 생략)...
"""

STEP2_USER_INSTRUCTION = """
당신은 학교 교육과정 분석 전문가이자 'Architect-Collaborator'입니다. 
제공된 요약 데이터를 기반으로 '학교 특색교육'과 '주요 교육과정' 리포트를 작성하십시오.
...(이하 출력 형식 및 예시 생략)...
"""

# ==========================================
# 3. 주요 기능 함수
# ==========================================
def get_model(temperature=0.0):
    return ChatGoogleGenerativeAI(
        model=MODEL_NAME,
        temperature=temperature,
        convert_system_message_to_human=True,
        max_retries=3,
        timeout=TIMEOUT_SECONDS
    )

def read_file(file_path):
    try:
        return file_path.read_text(encoding='utf-8')
    except UnicodeDecodeError:
        return file_path.read_text(encoding='cp949')

def process_step1_summarization():
    """1단계: input_data의 각 파일을 요약하여 input_processed에 저장"""
    PROCESSED_DIR.mkdir(parents=True, exist_ok=True)
    model = get_model(temperature=0.0)
    files = list(INPUT_DIR.glob("*.txt"))
    
    for file_path in files:
        processed_file = PROCESSED_DIR / f"summary_{file_path.name}"
        if processed_file.exists(): continue

        content = read_file(file_path)
        try:
            logging.info(f"요약 중: {file_path.name}")
            response = model.invoke([
                SystemMessage(content=STEP1_SYSTEM_PROMPT),
                HumanMessage(content=content)
            ])
            processed_file.write_text(response.content, encoding='utf-8')
            time.sleep(WAIT_TIME)
        except Exception as e:
            logging.error(f"오류: {e}")

def process_step2_final_analysis():
    """2단계: 요약된 파일들을 합쳐서 최종 리포트 생성"""
    OUTPUT_DIR.mkdir(parents=True, exist_ok=True)
    model = get_model(temperature=0.7)
    
    summary_files = list(PROCESSED_DIR.glob("*.txt"))
    combined_summary = "\n".join([f"[파일: {f.name}]\n{read_file(f)}" for f in summary_files])

    for i in range(2): # 2가지 버전 생성
        try:
            logging.info(f"최종 분석 결과 {i+1} 생성 중...")
            response = model.invoke([
                SystemMessage(content=STEP2_SYSTEM_PROMPT),
                HumanMessage(content=f"{STEP2_USER_INSTRUCTION}\n\n[데이터]:\n{combined_summary}")
            ])
            save_path = OUTPUT_DIR / f"특색교육 분석 결과 {i+1}.md"
            save_path.write_text(response.content, encoding='utf-8')
            time.sleep(WAIT_TIME)
        except Exception as e:
            logging.error(f"오류: {e}")

if __name__ == "__main__":
    process_step1_summarization()
    process_step2_final_analysis()

import os
import dotenv

dotenv.load_dotenv()

def get_env_variable(key: str) -> str:
    value = os.getenv(key)
    if value is None:
        raise ValueError(
            f"{key} 환경 변수가 설정되지 않았습니다. .env 파일을 확인해주세요."
        )
    return value

GEMINI_API_KEY = get_env_variable("GEMINI_API_KEY")
